sudo apt-get install libpq-dev
二、 配置.pc  libpq.pc
export PKG_CONFIG_PATH=/usr/pgsql-9.3/lib/pkgconfig
export PATH=/usr/pgsql-9.3/bin:$PATH
./configure --enable-core-pgsql-support
