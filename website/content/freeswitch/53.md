        if (ep) {
            ep->next = handle->last_sr_event;
        } else {
            handle->race_event = handle->last_sr_event;
        }

        handle->last_sr_event = NULL;

        esl_mutex_unlock(handle->mutex);
        esl_mutex_lock(handle->mutex);

        if (!handle->connected || handle->sock == ESL_SOCK_INVALID) {

            esl_mutex_unlock(handle->mutex);
            return ESL_FAIL;

        goto recv;
