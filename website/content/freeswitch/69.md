                                之前有群友说git中的编译不成功，我也折腾了一会，于是还是用files.freeswitch.org中的去用吧。
download:
wget http://files.freeswitch.org/downloads/libs/libvpx-1.4.1.tar.gz

tar zxvf libvpx-1.4.1.tar.gz

cd libvpx-1.4.1

./configure --enable-pic --disable-static --enable-shared

make

make install

cp /usr/local/lib/pkgconfig/vpx.pc /usr/lib64/pkgconfig/

libyuv:
 git clone https://freeswitch.org/stash/scm/sd/libyuv.git

 cd libyuv

 make -f linux.mk CXXFLAGS="-fPIC -O2 -fomit-frame-pointer -Iinclude/"

 make install

 cp /usr/lib/pkgconfig/libyuv.pc /usr/lib64/pkgconfig/
