<p>路由配置：</p>
<h1>dispatch request</h1>
<pre><code>route[DISPATCH] {
if(ds_is_from_list()) {
    route(SIPOUT);
} else {
    #!ifdef WITH_PATH
    if(!add_path_received()) {
            xlog(&#34;add_path_received failed\n&#34;);
    }
    #!endif
    if(is_method(&#34;REGISTER&#34;)) {
            $var(server_grp) = &#34;1&#34;;
    } else {
            $var(server_grp) = &#34;2&#34;;
    }
    if(!ds_select_dst($var(server_grp), &#34;0&#34;)) {
        send_reply(&#34;404&#34;, &#34;Dispatcher: No Destionation&#34;);
        exit;
    }
    route(RELAY);
}
exit;
}   
failure_route[MANAGE_FAILURE] {
    #route(NATMANAGE);

    if (t_is_canceled()) {
            exit;
    }

# try next dst - if only transcation failure of for 500 local timeout
if(t_check_status(&#34;500&#34;) or
    (t_branch_timeout() and !t_branch_replied())) {
    ds_mark_dst(&#34;ip&#34;);
    if(ds_next_dst()) {
        route(RELAY);
        exit;
    }
}
